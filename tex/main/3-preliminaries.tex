\section{Preliminaries}
\label{sec:preliminaries}

\subsection{Autoregressive Generation and Objective}

For the finite-horizon identities below, $T$ denotes a fixed deterministic horizon. Variable-length responses are represented by padding
with an absorbing end-of-sequence token and by zeroing padded action terms with the corresponding action mask. The post-termination
transition is deterministic and shared by both policies, so its likelihood ratio is $1$; this is the convention used whenever a common
horizon appears in a sum or product.

A (stochastic) policy $\pi$ generates a response $y = (y_1, \ldots, y_T)$ conditioned on a prompt $x$. The induced trajectory
probability admits the autoregressive factorization
\begin{equation*}
    P^{\pi}(y \mid x) = \prod_{t=1}^T \pi(y_t \mid x, y_{<t}).
\end{equation*}
For each position $t$, we define the context
\begin{equation*}
    c_t := (x, y_{<t}) = (x, \ylt).
\end{equation*}
The induced \emph{context visitation distribution} (the marginal distribution of length-$(t-1)$ prefixes, i.e. contexts before token $t$) is
\begin{equation*}
    d_t^{\pi}(c_t) = P(x)\, P^{\pi}(y_{<t} \mid x) = P(x) \prod_{s=1}^{t-1} \pi(y_s \mid c_s).
\end{equation*}
Given a scalar reward $R(x, y)$, the objective of the current policy $\pitheta$ is
\begin{equation*}
    J(\pitheta) = \E_{x \sim P(x),\, y \sim \pitheta(\cdot|x)}\big[R(x, y)\big].
\end{equation*}

\subsection{Notation}

\textbf{Policies.} We consider:
\begin{itemize}
    \item $\piroll$: the \emph{behavior} (rollout) policy used to generate samples; it is fixed during each policy update;
    \item $\pitheta$: the \emph{training} policy being optimized by gradient steps.
    \item $\piold$: the frozen PPO snapshot used by the clipped current-policy
    ratio; it is an implementation object and need not equal $\piroll$.
\end{itemize}
This notation follows long-horizon trust-region analyses for LLM-RL (e.g., \citet{li2025trm}).

\textbf{Importance ratios.} The per-token importance ratio and its log-ratio are
\begin{align}
    \rho_t
    &:= \frac{\pitheta(y_t \mid c_t)}{\piroll(y_t \mid c_t)},
    \label{eq:ratio}
    \\
    \log\rho_t
    &= \log \pitheta(y_t \mid c_t) - \log \piroll(y_t \mid c_t).
    \label{eq:log-ratio}
\end{align}

\textbf{Rollout-time anchoring.} In practice, inference and training stacks can
differ. The theoretical $\rho_t$ uses rollout-time cached log-probabilities
$\log \piroll(y_t\mid c_t)$. The implementation obtains the corresponding
correction through the factorization
$\rho_t^{\theta/\piroll}=r_t^{\theta/\piold}w_t^{\piold/\piroll}$;
see \Cref{app:mismatch} for practical mismatch modes and implementation
mappings.

\subsection{Divergence Measures}

We use token-level divergence measures between conditional token distributions, following the trust-region literature for long-horizon
LLM-RL~\citep{li2025trm}. For two policies $\pi$ and $\pi'$, define the token-level total variation and KL divergences at context $c_t$:

\begin{definition}[Token-Level Divergences]
For a context $c_t = (x, y_{<t})$ and two policies $\pi$ and $\pi'$,
\begin{align*}
    \Dtvtok(c_t;\pi,\pi')
    &:= \DTV\big(\pi(\cdot|c_t)\,\|\,\pi'(\cdot|c_t)\big)
    = \tfrac{1}{2}\sum_{v} \big|\pi(v|c_t) - \pi'(v|c_t)\big|, \\
    \Dkltok(c_t;\pi,\pi')
    &:= \DKL\big(\pi(\cdot|c_t)\,\|\,\pi'(\cdot|c_t)\big)
    = \sum_{v} \pi(v|c_t)\log\frac{\pi(v|c_t)}{\pi'(v|c_t)}.
\end{align*}
When the policy pair is clear from context, we write $\Dtvtok(c_t)$ and $\Dkltok(c_t)$.
\end{definition}

\begin{definition}[Max Token-Level Divergences and Expected Per-Position TV]
\label{def:max-div}
For the policy pair $(\piroll,\pitheta)$, define:
\begin{align*}
    \epsilon
    &:= \Dtvtokmax := \max_{t,\,c_t} \Dtvtok(c_t),
    \\
    \delta
    &:= \Dkltokmax := \max_{t,\,c_t} \Dkltok(c_t),
    \\
    \Dbar_t
    &:= \E_{c_t \sim d_t^{\piroll}}\!\big[\Dtvtok(c_t)\big].
\end{align*}
\end{definition}

\subsection{Assumptions}
\label{sec:assumptions}

The following assumptions are used throughout the paper:

\begin{assumption}[Mutual Policy Support]
\label{asm:support}
For all contexts $c_t$ and tokens $y_t$ in the vocabulary,
\begin{equation*}
    \piroll(y_t|c_t) > 0
    \quad\Longleftrightarrow\quad
    \pitheta(y_t|c_t) > 0.
\end{equation*}
That is, the two conditional policies have mutual support. This makes the per-token importance ratio
$\rho_t$ and its logarithm in Eqs.~\eqref{eq:ratio}--\eqref{eq:log-ratio} well-defined on every sampled token, and permits the
change-of-measure identities used below (in particular $\E_{\piroll}[\rho_t\mid c_t]=1$ and the TV/KL ratio formulas).
All ratio identities are interpreted on the rollout support
$\{y_t:\piroll(y_t\mid c_t)>0\}$. If both policies assign zero mass to a token,
the ratio at that token is assigned an arbitrary harmless value (the guarded
finite KL formalization uses $1$), because such tokens contribute zero to
rollout expectations and to the KL/TV sums. Proofs that work directly with
token ratios restrict them to the positive rollout support. Thus the assumption
does not require strict positivity on the whole vocabulary.
\end{assumption}

\begin{assumption}[Bounded Rewards]
There exists $R_{\max} < \infty$ such that $|R(x, y)| \le R_{\max}$ for all prompt-response pairs $(x, y)$.
\end{assumption}

\begin{assumption}[Finite Sequence Length]
\label{asm:finite-length}
There is a fixed finite generation cap $T$. Every response has at most $T$
active actions and is extended to this common horizon by the shared absorbing
EOS convention above.
\end{assumption}

\begin{remark}[Three-policy ratio decomposition in the implementation]
The mathematical statements below use the behavior/training pair
$(\piroll,\pitheta)$. The PPO implementation also has a frozen snapshot
$\piold$ and therefore uses two ratios:
\begin{equation*}
    r_t^{\theta/\piold}
    := \frac{\pitheta(y_t\mid c_t)}{\piold(y_t\mid c_t)},
    \qquad
    w_t^{\piold/\piroll}
    := \frac{\piold(y_t\mid c_t)}{\piroll(y_t\mid c_t)}.
\end{equation*}
On their common support,
\begin{equation*}
    \rho_t^{\theta/\piroll}
    := \frac{\pitheta(y_t\mid c_t)}{\piroll(y_t\mid c_t)}
    = r_t^{\theta/\piold} w_t^{\piold/\piroll}.
\end{equation*}
The corresponding log-probability implementation identity is checked by
\path{ImplementationBridge.current_rollout_ratio_factorization}; it is the
exponentiated form of the three-policy decomposition used below.
The prefix gate in the current implementation is computed from
$w_t^{\piold/\piroll}$, using cached rollout log-probabilities and the old-policy
log-probabilities; the PPO surrogate is computed from $r_t^{\theta/\piold}$.
When $\piold=\piroll$ at the beginning of an update, $w_t^{\piold/\piroll}=1$
and the PPO ratio equals the theoretical behavior-to-current ratio. For stale snapshots, the prefix results below
apply directly to the pair $(\piroll,\piold)$; transferring them to the
current pair additionally requires a bound on the PPO update.
\end{remark}

\subsection{The Surrogate Objective and Error Decomposition}

Samples are generated from a behavior policy $\piroll$ that generally differs from $\pitheta$. Following~\citet{kakade2002approximately},
the standard surrogate objective uses the per-step advantage under $\piroll$,
\begin{equation*}
    A_t^{\piroll}(c_t, y_t)
    := \E_{\piroll}\!\big[R(x, y)\mid c_t, y_t\big] - \E_{\piroll}\!\big[R(x, y)\mid c_t\big].
\end{equation*}
This yields the per-step surrogate
\begin{equation}
    \mathcal{L}_{\piroll}(\pitheta) = \sum_{t=1}^T \E_{c_t \sim d_t^{\piroll}}\!\Big[m_t(c_t)\,\E_{y_t \sim \piroll(\cdot|c_t)}\!\big[\rho_t \, A_t^{\piroll}(c_t, y_t)\big]\Big],
    \label{eq:surrogate}
\end{equation}
where $m_t(c_t)\in\{0,1\}$ is the predictable action mask from
\Cref{def:impl-surrogate}. In the unpadded fixed-horizon model $m_t\equiv1$;
with EOS padding it removes post-termination contexts, whose shared absorbing
transition has zero advantage and unit ratio.
The ratio $\rho_t$ is defined in Eq.~\eqref{eq:ratio}.

\begin{definition}[Trajectory-Level Surrogate]
\label{def:impl-surrogate}
Let $b$ be a baseline that is independent of the sampled trajectory $y$ conditional on the prompt $x$ (e.g., a prompt-level baseline
computed from other independent rollout samples in the same prompt group). The rollout policy and the baseline sampling law are fixed
during the update. Let $m_t=m_t(c_t)\in\{0,1\}$ denote the predictable action mask: it equals one while generation is active, including
the action that emits EOS, and zero at subsequent padding positions. Define the trajectory-level advantage
\begin{equation*}
    A := R(x, y) - b.
\end{equation*}
The unclipped, unnormalized trajectory-level surrogate underlying critic-free implementations is
\begin{equation*}
    \tilde{\mathcal{L}}_{\piroll}(\pitheta) := \E_{\piroll}\!\Big[A \cdot \sum_{t=1}^T m_t\rho_t\Big].
\end{equation*}
\end{definition}

\begin{proposition}[Exact Centered Surrogate Equivalence]
\label{prop:surrogate-equivalence}
For finite prompt, action, and continuation supports, under mutual policy support and the finite/EOS-padded horizon convention, assume that the expectations below exist and that $b$ and
$m_t$ satisfy \Cref{def:impl-surrogate}. Then, for any current policy $\pitheta$,
\begin{equation}
    \tilde{\mathcal L}_{\piroll}(\pitheta)
    -\tilde{\mathcal L}_{\piroll}(\piroll)
    =\mathcal L_{\piroll}(\pitheta)
    =\E_{\piroll}\!\left[R(x,y)\sum_{t=1}^{T}m_t(\rho_t-1)\right].
    \label{eq:surrogate-equivalence}
\end{equation}
The subtracted quantity is constant during the policy update; the equality does not require $\pitheta=\piroll$.
\end{proposition}

The proof is in \Cref{app:surrogate-equivalence}. This is a population
identity; individual Monte Carlo estimates need not coincide. The random-length
batch reductions used by ProMax are retained in \Cref{sec:promax-certificate}.

\begin{remark}[Gradient Interpretation and ProMax Modifications]
Where differentiable, Eq.~\eqref{eq:surrogate-equivalence} implies equality of the gradients of the two surrogates while keeping the
rollout distribution fixed. At the on-policy evaluation point, the REINFORCE identity gives
\begin{equation*}
    \E_{\pitheta}\!\Big[A \cdot \sum_{t=1}^T m_t\nabla_\theta \log \pitheta(y_t \mid c_t)\Big] = \nabla_\theta J(\pitheta),
\end{equation*}
under the usual score-function differentiation conditions. Away from that point, equality of the two surrogate gradients does not
equate either gradient with $\nabla_\theta J(\pitheta)$.
The actual ProMax loss additionally applies adaptive normalization, an action-dependent prefix gate, PPO clipping, and the selected
batch reduction. These operations require their own analysis; the exact finite-batch decomposition in
\Cref{prop:promax-objective-split} retains all of them. In particular, detaching a prefix gate does not make it independent of the sampled
action.
\end{remark}

The approximation error of the per-step surrogate is
\begin{equation*}
    \mathrm{Error}(\pitheta) := J(\pitheta) - J(\piroll) - \mathcal{L}_{\piroll}(\pitheta).
\end{equation*}
Under the finite-horizon Performance Difference Identity~\citep{kakade2002approximately},
and assuming finite context/token supports so that the conditional expectations are finite sums,
this error decomposes as:
\begin{equation}
    |\mathrm{Error}(\pitheta)| \le \sum_{t=1}^T 2\|g_t\|_\infty \cdot \|d_t^{\pitheta} - d_t^{\piroll}\|_{\mathrm{TV}},
    \label{eq:error-decomp}
\end{equation}
where
\begin{equation*}
    g_t(c_t) := m_t(c_t)\,\E_{y_t \sim \pitheta(\cdot\mid c_t)}\!\big[A_t^{\piroll}(c_t, y_t)\big]
\end{equation*}
is the expected advantage shift. Classical bounds on Eq.~\eqref{eq:error-decomp} scale as $O(T^2)$~\citep{kakade2002approximately,
achiam2017constrained}, rendering them theoretically vacuous for long-horizon LLM tasks with responses of thousands of tokens. Recent
work~\citep{li2025trm} derives tighter bounds,
including the Pinsker-Marginal bound ($O(T^{3/2})$), the Mixed bound ($O(T)$), and the data-dependent Adaptive bound.

The finite-sum error derivation and its formalization are recorded in
\Cref{app:surrogate-equivalence}.

The Adaptive bound of \citet{li2025trm} is particularly relevant to our work.
We give a complete finite-tree proof, including the linear future-TV route,
to fix its constants and connect it to the same surrogate used by the method.

\begin{theorem}[Finite-tree Adaptive remainder bound]
\label{thm:adaptive-bound}
For a finite prompt distribution and finite vocabulary, let both policies be
normalized history-dependent kernels satisfying \Cref{asm:support} and the
fixed/EOS-padded horizon convention. Suppose $|R|\le R_{\max}$ with
$R_{\max}\ge0$, and let $\epsilon,\delta$ bound the local TV and reverse KL
at every context before horizon $T$, as in \Cref{def:max-div}. Then
\begin{equation}
    B_{\mathrm{Adap}}^{*} = 4 R_{\max} \sum_{t=1}^{T} \Dbar_t \cdot \min\!\Big(1,\;(T\!-\!t)\,\epsilon,\;\sqrt{\tfrac{(T-t)\,\delta}{2}}\Big),
    \label{eq:adaptive-bound}
\end{equation}
satisfies
\begin{equation*}
    |J(\pitheta)-J(\piroll)-\mathcal L_{\piroll}(\pitheta)|
    \le B_{\mathrm{Adap}}^{*},
    \qquad
    J(\pitheta)-J(\piroll)
    \ge \mathcal L_{\piroll}(\pitheta)-B_{\mathrm{Adap}}^{*}.
\end{equation*}
The same bound controls the absolute reward-weighted trajectory remainder in
\Cref{def:llm-surrogate}.
\end{theorem}

\begin{proof}
The exact history-tree error recursion separates the current token's
probability shift from the difference in future expected rewards.
The latter is bounded by $2R_{\max}$ times the future trajectory TV, which is
at most each of $1$, $(T-t)\epsilon$, and $\sqrt{(T-t)\delta/2}$.
Summing the local interaction under rollout prefixes gives
Eq.~\eqref{eq:adaptive-bound}. The full recursion and finite prompt averaging
are proved in \Cref{app:adaptive-proof} and checked in
\path{formal/AdaptiveBound.lean}.
\end{proof}

The factor $R_{\max}$ is required for general bounded rewards; $R_{\max}=1$
recovers the normalized-reward form. This result retains the per-position
expected TV and the remaining future horizon. It motivates causal prefix
filtering, while the sampled gate's own guarantees are stated separately in
\Cref{thm:pre-causal-mechanism}; a sampled acceptance event does not establish
the population TV/KL assumptions of \Cref{thm:adaptive-bound}.

\subsection{Existing Importance Sampling Correction Methods}
\label{sec:existing-is}

Several methods have been proposed to stabilize policy-gradient updates by constraining the per-token importance ratio $\rho_t$ between the
training policy $\pitheta$ and the behavior policy $\piroll$. We use the per-token log-ratio $\log\rho_t$ defined in
Eq.~\eqref{eq:log-ratio}.

\noindent Table~\ref{tab:is-methods} summarizes these stabilization mechanisms under a unified notation.
\begin{table}[ht]
\centering
\small
\caption{Existing trust-region and IS stabilization methods in critic-free LLM-RL.}
\label{tab:is-methods}
\renewcommand{\arraystretch}{1.2}
\setlength{\tabcolsep}{4pt}
\begin{tabular}{cccc}
\toprule
\textbf{Method} & \textbf{IS Statistic} & \textbf{Truncation} & \textbf{Granularity} \\
\midrule
TIS & $\rho_t$ & $w_t^{\mathrm{tok}}=\mathrm{clip}(\rho_t,1-\epslow,1+\epshigh)$ & Token \\
IcePop & $\rho_t$ & $M_t^{\mathrm{tok}}=\mathbb{I}[\rho_t \in [1-\epslow,1+\epshigh]]$ & Token \\
TRM & $\rho_t$ & $M^{\mathrm{TRM}}(x,y)=\mathbb{I}\!\big[\max_t\,\Dkltok(c_t;\piroll,\pitheta)\le \delta_{\mathrm{TRM}}\big]$ & Sequence \\
GSPO & $\rhoseq$ & $w^{\mathrm{seq}}=\mathrm{clip}(\rhoseq,1-\epslow,1+\epshigh)$ & Sequence \\
\bottomrule
\end{tabular}
\end{table}

The table uses the theoretical behavior-to-training ratio for a common
notation. In the PPO implementation, the corresponding mismatch correction
weights are evaluated from $w_t^{\piold/\piroll}$, while the differentiable
PPO ratio is $r_t^{\theta/\piold}$, as specified above.

\noindent Here $\rhoseq$ is the length-normalized sequence likelihood ratio (a geometric mean of token ratios),
\begin{equation*}
    \begin{aligned}
        \rhoseq
        &:= \left(\frac{P^{\pitheta}(y \mid x)}{P^{\piroll}(y \mid x)}\right)^{\frac{1}{T}}
        \\
        &= \left(\prod_{t=1}^{T}\rho_t\right)^{\frac{1}{T}}
        = \exp\!\left(\frac{1}{T}\sum_{t=1}^{T}\log\rho_t\right),
    \end{aligned}
\end{equation*}
where $T$ is the fixed (or EOS-padded) horizon used for the comparison. For an unpadded response, this reduces to its actual length
$T=|y|$. This definition matches the sequence-level ratio used in GSPO~\citep{gspo2025} (denoted $s_i(\theta)$ in the GSPO paper) up to
notation.

TRM (Trust Region Masking, \citet{li2025trm}) uses a binary \emph{sequence-level} mask $M^{\mathrm{TRM}}(x,y)$ to discard trajectories that violate the
trust region. A standard (max-based) criterion is
\begin{equation*}
    M^{\mathrm{TRM}}(x,y)
    := \mathbb{I}\!\Big[\max_{t}\,\Dkltok(c_t;\piroll,\pitheta) \le \delta_{\mathrm{TRM}}\Big].
\end{equation*}
A comparative discussion of these mechanisms and their limitations is provided in \Cref{sec:unified}.

\begin{remark}[Context Shift Propagation]
\label{rem:coupling}
Fix a prompt and a finite vocabulary, with normalized policies on complete
prefix contexts. Let $s$ be the first position of disagreement: the policies
agree at every rollout-reachable context before $s$, and
$\Dtvtok(c_s;\pi,\pi')>0$ at some context with $d_s^\pi(c_s)>0$. Then
\begin{equation*}
    \|d_{s+1}^{\pi}-d_{s+1}^{\pi'}\|_{\mathrm{TV}}
    = \E_{c_s\sim d_s^\pi}[\Dtvtok(c_s;\pi,\pi')] > 0.
\end{equation*}
Moreover, full-prefix TV is nondecreasing with prefix length, so every later
context distribution retains at least this discrepancy. Marginalizing the
last token recovers the preceding complete prefix; the triangle inequality
therefore prevents its TV from increasing under that marginalization.
No support restriction on subsequent policy kernels is needed, including
after a shared absorbing EOS transition. The equality, monotonicity, and
first-disagreement implication are checked by
\path{prefixTV_next_of_equal_prefix_mass}, \path{prefixTV_mono}, and
\path{first_disagreement_propagates} in \path{formal/TreeTV.lean}.
This statement concerns full histories; a representation that discards
earlier tokens need not retain their discrepancy. See also \citet{li2025trm}.
\end{remark}
